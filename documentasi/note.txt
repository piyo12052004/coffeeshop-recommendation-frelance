docker-compose
services:
  coffeeshop-app:
    container_name: coffeeshop-app

    build:
      context: .
      dockerfile: Dockerfile

    image: coffeeshop-recommendation:latest

    restart: unless-stopped

    ports:
      - "5000:5000"

    environment:
      DB_HOST: mysql-aset
      DB_PORT: 3306
      DB_DATABASE: coffeeshop_project
      DB_USERNAME: root
      DB_PASSWORD: root123

    tty: true
    stdin_open: true

    networks:
      - coffeeshop-network


networks:
  coffeeshop-network:
    driver: bridge

step 4 
build file dockerfile dan docker-compose

step 5 
hubungkan network mysql-aset dengan coffeeshop-network agar saling terhubung 
docker inspect coffeeshop-app --format='{{json .NetworkSettings.Networks}}'
dan:
docker inspect mysql-aset --format='{{json .NetworkSettings.Networks}}'
Pastikan keduanya punya:
coffeeshop-recommendation_coffeeshop-network

step 6 
masuk ke mysql-aset mengunakan exec buat database coffeeshop_project dan import database
